Теория 1

Цифра, число, конец числа

Три приёма, на которых держатся все задачи блока:

  1. Цифра — символ с кодом от 48 до 57, её значение — code - 48.
  2. Число из цифр — слева направо: number = number * 10 + digit.
  3. Конец числа — первая не-цифра после цифр. Чтобы его заметить, нужен флаг started. Цикл до i <= str.length делает лишний шаг за концом строки (там charCodeAt даёт NaN), и число в самом конце тоже «заканчивается».
const s = 'Paid 750, left 14690';
let number = 0;
let started = false;
for (let i = 0; i <= s.length; i++) {
  const code = s.charCodeAt(i);
  if (code >= 48 && code <= 57) {
    number = number * 10 + (code - 48);
    started = true;
  } else if (started) {
    console.log('число закончилось:', number);
    number = 0;
    started = false;
  }
}

function digitsToNumber(str) {

Теория 2

Дробь: все цифры в целое, делим один раз

Точку (или запятую) в число не добавляем — только запоминаем, что она была. Все цифры идут в одно целое number, а каждая цифра после разделителя умножает делитель scale на 10. В конце — одно деление: так нет ошибок округления.

Разделитель считается частью числа, только если после него цифра. Иначе это конец предложения ('USDT.') или запятая в перечислении ('12, 13').

const s = '12,50 USDT.';
let number = 0;
let scale = 1;
let afterSep = false;
for (let i = 0; i < s.length; i++) {
  const code = s.charCodeAt(i);
  const next = s.charCodeAt(i + 1);
  if (code >= 48 && code <= 57) {
    number = number * 10 + (code - 48);
    if (afterSep) scale *= 10;
  } else if ((code === 44 || code === 46) && !afterSep && next >= 48 && next <= 57) {
    afterSep = true;
  }
}
console.log(number, scale, number / scale);   // 1250 100 12.5

Напиши parseSimpleFloat(str) — строка состоит из цифр и не больше одной точки. Собери число способом «слева направо, делим один раз».

Показать кодСкрыть
function parseSimpleFloat(str) {
  let number = 0;
  let scale = 1;
  let afterDot = false;

  for (let i = 0; i < str.length; i++) {
    const code = str.charCodeAt(i);
    if (code === 46) {
      afterDot = true;
      continue;
    }
    number = number * 10 + (code - 48);
    if (afterDot) scale *= 10;
  }
  return number / scale;
}

console.log(parseSimpleFloat('3.14')); // 3.14
console.log(parseSimpleFloat('10'));   // 10
Разбор
  1. Все цифры — и до точки, и после — собираем в одно целое число number по формуле Горнера.
  2. Точку не добавляем в число, а только поднимаем флаг afterDot.
  3. Каждая цифра после точки умножает делитель scale на 10. Две цифры после точки — делитель 100.
  4. В конце — одно деление: 314 / 100 = 3.14.

Для строки без точки scale так и останется 1, и деление ничего не испортит. Тест '9.95' — тот самый случай, на котором ломается «прибавление по цифре».

function toLowerCode(code) {
  return code >= 65 && code <= 90 ? code + 32 : code;
}

function isWordAt(str, pos, word) {
  for (let k = 0; k < word.length; k++) {
    if (toLowerCode(str.charCodeAt(pos + k)) !== word.charCodeAt(k)) return false;
  }
  return true;
}

function countBalanceDecimal(str) {
  let number = 0;
  let scale = 1;
  let started = false;
  let afterDot = false;
  let spent = 0;
  let balance = 0;
  let isBalance = false;

  for (let i = 0; i <= str.length; i++) {
    const code = str.charCodeAt(i);
    const next = str.charCodeAt(i + 1);

    if (isWordAt(str, i, 'balance')) isBalance = true;

    if (code >= 48 && code <= 57) {
      number = number * 10 + (code - 48);
      if (afterDot) scale *= 10;
      started = true;

      afterDot = true;
    } else if (started) {
      const value = number / scale;
      if (isBalance) {
        balance = value;
        isBalance = false;
      } else {

      }
      number = 0;
      scale = 1;
      started = false;

function allAmounts(message) {
  const result = [];
  let number = 0;
  let scale = 1;
  let started = false;
  let afterSep = false;
  let negative = false;

  for (let i = 0; i <= message.length; i++) {
    const code = message.charCodeAt(i);
    const next = message.charCodeAt(i + 1);
    const nextIsDigit = next >= 48 && next <= 57;

    if (code >= 48 && code <= 57) {
      if (!started) negative = message.charCodeAt(i - 1) === 45;
      number = number * 10 + (code - 48);
      if (afterSep) scale = scale * 10;
      started = true;
    } else if ((code === 44 || code === 46) && started && !afterSep && nextIsDigit) {
      afterSep = true;
    } else if (started) {
      result[result.length] = (negative ? -number : number) / scale;
      number = 0;
      scale = 1;
      started = false;
      afterSep = false;
      negative = false;
    }
  }
  return result;
}

Разбор

Задача с собеседования, в которой break заменён на сброс — как parseLastBalance, только каждое число не перезаписывается, а складывается в массив.

Теория 3

Справа налево: вес цифры

Когда идём с конца, первая встреченная цифра — младший разряд. Каждая цифра умножается на вес power (1, 10, 100…), вес растёт на каждом шаге.

Встретили разделитель — значит, всё, что уже собрано, было дробной частью. Текущий вес и есть делитель: scale = power.

const s = '56,02';
let number = 0;
let power = 1;
let scale = 1;
for (let i = s.length - 1; i >= 0; i--) {
  const code = s.charCodeAt(i);
  if (code === 44 || code === 46) {
    scale = power;
    continue;
  }
  number += (code - 48) * power;
  power *= 10;
}
console.log(number, scale, number / scale);   // 5602 100 56.02

После цикла power — это 10 в степени «сколько цифр прочитано». Это пригодится в последней задаче экзамена.

function parseFromEnd(str) {
  let number = 0;
  let power = 1;
  for (let i = str.length - 1; i >= 0; i--) {
    number += (str.charCodeAt(i) - 48) * power;
    power *= 10;
  }
  return number;
}

console.log(parseFromEnd('4096')); // 4096
Разбор

Справа налево каждую цифру умножаем на её вес — power: единицы (1), десятки (10), сотни (100)… После каждой цифры вес растёт в 10 раз.

Как собирается '4096' с конца
'6' × 1     → 6
'9' × 10    → 96
'0' × 100   → 96
'4' × 1000  → 4096

Этот способ понадобится, когда число надо найти в конце строки: начинаем с конца, и первая встреченная цифра — это сразу разряд единиц.

      number = 0;
      scale = 1;

function contactNumber(str) {
  let result = 0;    // все цифры результата подряд, как одно целое
  let scale = 1;     // на что поделить в конце (10 в степени «цифр после запятой»)
  let count = 0;     // сколько чисел уже собрано
  let current = 0;   // число, которое читаем сейчас
  let power = 1;     // вес следующей цифры: 1, 10, 100…

  for (let i = str.length - 1; i >= 0; i--) {
    const code = str.charCodeAt(i);

    if (code >= 48 && code <= 57) {
      current += (code - 48) * power;
      power *= 10;

      // продолжается ли число левее?
      const left = str.charCodeAt(i - 1);
      const left2 = str.charCodeAt(i - 2);
      const leftIsDigit = left >= 48 && left <= 57;
      const leftIsSep = (left === 44 || left === 46) && left2 >= 48 && left2 <= 57;

      if (!leftIsDigit && !leftIsSep) {          // число закончилось
        if (count === 0) {
          result = current;                      // первое справа — начало результата
        } else {
          result = result * power + current;     // остальные — дописываем в конец
          if (scale > 1) scale *= power;         // если была дробь, новые цифры тоже после запятой
        }
        count++;
        current = 0;
        power = 1;
      }
    } else if ((code === 44 || code === 46) && power > 1 && count === 0) {
      scale = power;                             // разделитель внутри первого числа
    }
  }
  return count === 0 ? null : result / scale;
}

console.log(contactNumber('Right part 72 is 49 asdhsad 15 and left part is 56,02')); // 56.02154972
Разбор

Формулировка на экзамене путает («первое справа — right part»), поэтому ориентируйся на примеры: первое число справа — начало ответа, остальные числа приклеиваются к нему цифрами. '... 72 ... 49 ... 15 ... 56,02' → 56,02 + 15 + 49 + 72 = 56.02154972.

Всё решение — три переменные на «текущее число» и три на «ответ»:

  1. Текущее число читаем способом «задом наперёд»: current += цифра * power, power *= 10. Разделитель внутри первого числа запоминает делитель scale = power — как в parseFloatFromEnd.
  2. Конец числа при обратном проходе — когда левее нет ни цифры, ни разделителя с цифрой перед ним. Поэтому смотрим на i - 1 и i - 2. У индекса -1 charCodeAt даёт NaN — это «не цифра», и число в самом начале строки тоже закроется. Лишний шаг цикла не нужен (и нельзя — цикл задан условием).
  3. Приклеить число к ответу: result * power + current. После чтения числа power равен 10 в степени «сколько в нём цифр» — ровно на столько разрядов надо сдвинуть ответ.
  4. Если у первого числа была дробь, каждая приклеенная цифра тоже стоит после запятой: scale *= power. Если дроби не было (5602), scale остаётся 1, и цифры просто удлиняют целое: 56021493.

Деление одно, в самом конце: 5602154972 / 100000000 = 56.02154972.

Напиши invert(obj) — поменять ключи и значения местами. Если значения повторяются, побеждает последний ключ.

invert({ a: 'x', b: 'y' }) → { x: 'a', y: 'b' }.

Показать кодСкрыть
function invert(obj) {
  const result = {};
  for (const [key, value] of Object.entries(obj)) result[value] = key;
  return result;
}

console.log(invert({ a: 'x', b: 'y' })); // { x: 'a', y: 'b' }
Разбор

Object.entries даёт пары, а деструктуризация [key, value] сразу раскладывает пару на две переменные. Значение становится ключом — и превращается в строку: ключи объекта бывают только строками (и символами). Поэтому в { a: 1, b: 1 } оба ключа пишут в одно и то же '1', и остаётся последний.

3Легко

Напиши getOr(obj, key, fallback) — значение своего свойства key, а если такого свойства нет — fallback. Свойство со значением undefined — есть, его значение и возвращаем.

getOr({ a: 0 }, 'a', 5) → 0, getOr({}, 'a', 5) → 5.

Показать кодСкрыть
function getOr(obj, key, fallback) {
  return Object.hasOwn(obj, key) ? obj[key] : fallback;
}

console.log(getOr({ a: 0 }, 'a', 5)); // 0
Разбор

Три способа спросить «есть ли свойство», и только один подходит:

  • obj[key] ?? fallback — ошибается на { a: undefined }: свойство есть, а ответ fallback;
  • key in obj — смотрит и в прототип: 'toString' in {} — это true;
  • Object.hasOwn(obj, key) — только свои свойства, независимо от значения. Это то, что нужно.

Теория 2

Приём: entries → методы массива → fromEntries

У объекта нет map и filter. Но можно превратить его в массив пар, обработать методами массива и собрать обратно.

const prices = { apple: 30, milk: 45, bread: 20 };
const cheap = Object.fromEntries(
  Object.entries(prices).filter(([name, price]) => price < 40)
);
console.log(cheap);   // { apple: 30, bread: 20 }

Этот приём решает почти все задачи вида «сделай из объекта другой объект».

Напиши filterObject(obj, predicate) — новый объект только из тех свойств, для которых predicate(value, key) истинно.

7Средне★ минимум

Напиши pick(obj, keys) — новый объект только с перечисленными ключами (если такого ключа в объекте нет — пропустить).

pick({ a: 1, b: 2, c: 3 }, ['a', 'c']) → { a: 1, c: 3 }.

Показать кодСкрыть
function pick(obj, keys) {
  const result = {};
  for (const key of keys) {
    if (Object.hasOwn(obj, key)) result[key] = obj[key];
  }
  return result;
}

console.log(pick({ a: 1, b: 2, c: 3 }, ['a', 'c'])); // { a: 1, c: 3 }
Разбор

Идём не по объекту, а по списку нужных ключей — так быстрее, если объект большой, а ключей нужно мало. Object.hasOwn отличает «ключа нет» от «ключ есть, но значение undefined».

Обратная задача omit (выбросить ключи) удобнее через filterObject: оставить те, которых нет в списке.

8Сложно

Напиши diffObjects(a, b) — что изменилось между двумя плоскими объектами: { added, removed, changed } — массивы ключей. removed и changed — в порядке ключей a, added — в порядке ключей b. Значения сравниваются через ===.

diffObjects({ a: 1, b: 2, c: 3 }, { a: 1, b: 5, d: 4 }) → { added: ['d'], removed: ['c'], changed: ['b'] }.

Показать кодСкрыть
function diffObjects(a, b) {
  const result = { added: [], removed: [], changed: [] };
  for (const key of Object.keys(a)) {
    if (!Object.hasOwn(b, key)) result.removed.push(key);
    else if (a[key] !== b[key]) result.changed.push(key);
  }
  for (const key of Object.keys(b)) {
    if (!Object.hasOwn(a, key)) result.added.push(key);
  }
  return result;
}

console.log(diffObjects({ a: 1, b: 2, c: 3 }, { a: 1, b: 5, d: 4 }));
Разбор

Два прохода: по ключам a находим удалённые и изменённые, по ключам b — добавленные. Ключ, который есть в обоих объектах, проверяется только в первом проходе.

Object.hasOwn снова важнее проверки на undefined: в третьем тесте ключ x был (пусть и со значением undefined), а потом исчез — это удаление.

Теория 3

Копии: поверхностная и глубокая

{ ...obj } и Object.assign({}, obj) копируют только верхний уровень. Вложенные объекты и массивы остаются общими — это одна и та же ссылка.

Глубокая копия:

  • structuredClone(obj) (Node 17+) — копирует всё, включая Date, Map, Set и циклические ссылки; функции копировать не умеет — бросает ошибку;
  • JSON.parse(JSON.stringify(obj)) — старый способ, но с потерями: undefined и функции пропадают, Date становится строкой, NaN — null.
const a = { name: 'Kate', tags: ['js'] };
const b = { ...a };
b.name = 'Max';
b.tags.push('css');
console.log(a.name, a.tags);   // Kate [ 'js', 'css' ]

const c = structuredClone(a);
c.tags.push('html');
console.log(a.tags.length, c.tags.length);   // 2 3

Напиши defaults(target, ...sources) — дописать в target значения из источников, но только для ключей, где в target сейчас undefined. Источники проверяются по порядку. Вернуть target.

defaults({ a: 1 }, { a: 2, b: 2 }, { b: 3, c: 3 }) → { a: 1, b: 2, c: 3 }.

Показать кодСкрыть
function defaults(target, ...sources) {
  for (const src of sources) {
    for (const key of Object.keys(src)) {
      if (target[key] === undefined) target[key] = src[key];
    }
  }
  return target;
}

console.log(defaults({ a: 1 }, { a: 2, b: 2 }, { b: 3, c: 3 })); // { a: 1, b: 2, c: 3 }
Разбор

Противоположность Object.assign: там побеждает последний источник, здесь — первое найденное значение, а своё значение target вообще не трогается. null — это настоящее значение («пусто намеренно»), поэтому его не заменяем.

...sources — rest-параметр: сколько бы источников ни передали, они придут массивом.

Напиши deepCopy(value) — глубокую копию: примитивы возвращаются как есть, массивы и обычные объекты копируются на всех уровнях, Date — новой датой с тем же временем.

deepCopy({ list: [1, { x: 2 }] }) — новый объект, и list[1] в нём — тоже новый объект.

Показать кодСкрыть
function deepCopy(value) {
  if (value === null || typeof value !== 'object') return value;
  if (value instanceof Date) return new Date(value.getTime());

  const copy = Array.isArray(value) ? [] : {};
  for (const key of Object.keys(value)) copy[key] = deepCopy(value[key]);
  return copy;
}

const src = { d: new Date(0), list: [1, { x: 2 }] };
const cp = deepCopy(src);
console.log(cp.list[1] !== src.list[1], cp.d.getTime()); // true 0
Разбор

Рекурсия здесь естественна: «копия объекта» — это новый объект, где каждое значение — копия значения. Три случая:

  1. примитив (и null) — копировать нечего, вернуть как есть;
  2. Date — объект, но без своих ключей: Object.keys(new Date()) пустой, поэтому дату копируем отдельно;
  3. массив или объект — новый контейнер того же вида и рекурсия по ключам.

Чего эта версия не умеет: циклические ссылки (зациклится) и Map/Set. Для этого есть structuredClone, а свой вариант с циклами — в блоке 11 через WeakMap.

Теория 4

freeze, seal и настройки свойства

  • Object.freeze(obj) — свойства нельзя менять, добавлять и удалять. Заморозка поверхностная: вложенные объекты остаются изменяемыми.
  • Object.seal(obj) — нельзя добавлять и удалять, но менять можно.
  • Object.preventExtensions(obj) — нельзя только добавлять.

В обычном скрипте запрещённое действие молча игнорируется, а в строгом режиме ('use strict', модули, классы) бросает TypeError.

У каждого свойства есть настройки: writable (можно менять), enumerable (видно в Object.keys, for...in, JSON.stringify), configurable (можно удалять и перенастраивать). У обычного свойства все три true, а у созданного через Object.defineProperty по умолчанию — false.

const cfg = Object.freeze({ port: 80, db: { host: 'a' } });
cfg.port = 3000;
cfg.db.host = 'b';
console.log(cfg.port, cfg.db.host);   // 80 b

Напиши addHidden(obj, key, value) — добавить свойство, которое можно прочитать, но которого не видно в Object.keys и JSON.stringify и которое нельзя изменить. Вернуть obj.

Напиши deepFreeze(obj) — заморозить объект на всех уровнях (все вложенные объекты и массивы тоже). Без рекурсии: используй стек. Объект может ссылаться сам на себя. Вернуть obj.

Показать кодСкрыть
function deepFreeze(obj) {
  const stack = [obj];                 // объекты, которые ещё надо заморозить
  while (stack.length > 0) {
    const current = stack.pop();
    Object.freeze(current);
    for (const key of Object.keys(current)) {
      const value = current[key];
      if (typeof value === 'object' && value !== null && !Object.isFrozen(value)) {
        stack.push(value);
      }
    }
  }
  return obj;
}

const o = deepFreeze({ a: { b: { c: 1 } }, list: [1, { x: 2 }] });
o.a.b.c = 100;
console.log(o.a.b.c); // 1
Разбор

Рекурсивная версия короче: заморозить объект и вызвать себя для каждого вложенного. Но на экзамене просили «без рекурсии» — поэтому приём, который понадобится и для flat, и для JSON.stringify: стек задач.

  1. В стеке лежат объекты, которые ещё не заморожены.
  2. Берём верхний, замораживаем, все его вложенные объекты кладём в стек.
  3. Стек пуст — всё заморожено.

Проверка !Object.isFrozen(value) защищает от циклов: объект, который ссылается сам на себя, после заморозки второй раз в стек не попадёт, и цикл закончится.

Поиск: indexOf, includes, find

  • indexOf и lastIndexOf сравнивают через ===, поэтому не находят NaN (NaN !== NaN).
  • includes сравнивает по правилу SameValueZero: как ===, но NaN равен NaN. Это правило в JS используют ещё Map и Set.
  • find и findIndex ищут по условию-функции; findLast и findLastIndex — с конца.
  • some на пустом массиве — false (никто не подошёл), every — true (никто не нарушил).

Своё сравнение SameValueZero: x === y || (x !== x && y !== y) — второе условие верно только для двух NaN: это единственное значение, не равное самому себе.

function myIncludes(arr, value, fromIndex = 0) {
  let start = fromIndex < 0 ? arr.length + fromIndex : fromIndex;
  if (start < 0) start = 0;

  for (let i = start; i < arr.length; i++) {
    const x = arr[i];
    if (x === value || (x !== x && value !== value)) return true;   // SameValueZero
  }
  return false;
}

  2. Откуда искать. Отрицательный fromIndex считается от конца: -1 — последний элемент. Если ушли левее нуля (-10 на массиве из трёх), ищем с начала.

function partition(arr, predicate) {
  const yes = [];
  const no = [];
  for (const item of arr) {
    if (predicate(item)) yes.push(item);
    else no.push(item);

Теория 1

Как работает настоящий join

  1. Разделитель: не передан (undefined) — запятая; иначе превращается в строку (join(0) склеит через '0').
  2. Для каждого индекса от 0 до length - 1: перед всеми, кроме первого, — разделитель.
  3. Элемент null, undefined или дырка — пустая строка. Остальное — String(элемент); вложенный массив при этом сам склеится через запятую.
console.log([1, null, undefined, , 'a'].join('-'));   // 1----a
console.log([1, [2, [3]]].join(';'), [1, 2].join(), [1, 2].join(0));   // 1;2,3 1,2 102

Как сделать метод своим для всех массивов — записать функцию в прототип; внутри массив доступен как this:

Array.prototype.myJoin = function (separator) {
  return myJoin(this, separator);
};
[1, 2].myJoin('+');   // '1+2'

Напиши myJoin(arr, separator) — копию arr.join(separator) 1 в 1, одним циклом, без join.

function myJoin(arr, separator) {
  const sep = separator === undefined ? ',' : String(separator);
  let result = '';
  for (let i = 0; i < arr.length; i++) {
    if (i > 0) result += sep;
    const el = arr[i];                       // у дырки тоже undefined
    if (el !== null && el !== undefined) result += String(el);
  }
  return result;
}

console.log(myJoin([1, null, undefined, , 'a'], '-')); // '1----a'
Разбор

Три правила из теории — три строки кода:

  1. sep — запятая только если разделителя нет; 0 или '' — настоящие разделители, поэтому проверяем именно === undefined, а не !separator.
  2. Разделитель ставим перед каждым элементом, кроме первого, — так нет лишнего разделителя в конце.
  3. null и undefined дают пустую строку, но разделитель вокруг них всё равно ставится: '1----a' — четыре дефиса на пять элементов.

String([2, 3]) вызывает у массива toString, а тот — встроенный join(','). Так работает и настоящий join: он не знает про вложенность, просто превращает элемент в строку.

Теория 2

Как работает настоящий reduce

arr.reduce(callback, initialValue) проходит массив и копит одно значение — аккумулятор. callback получает (acc, элемент, индекс, массив) и возвращает новый acc.

  1. Начальное значение передано (важно: передано, даже если это undefined) — acc = оно, обход с индекса 0.
  2. Не передано — acc = первый существующий элемент, обход со следующего.
  3. Дырки пропускаются.

function myReduce(arr, callback, initialValue) {
  if (typeof callback !== 'function') throw new TypeError(callback + ' is not a function');

  const len = arr.length;               // длина — один раз, в начале
  let hasAcc = arguments.length > 2;    // начальное значение передали?
  let acc = initialValue;

  for (let i = 0; i < len; i++) {
    if (!(i in arr)) continue;          // дырка
    if (!hasAcc) {                      // первый существующий элемент становится acc
      acc = arr[i];
      hasAcc = true;
      continue;
    }
    acc = callback(acc, arr[i], i, arr);
  }

  if (!hasAcc) throw new TypeError('Reduce of empty array with no initial value');
  return acc;
}

Свой flat без рекурсии

С рекурсией flat пишется в пять строк: встретил массив — вызови себя. Но на экзамене рекурсия запрещена, а циклов можно не больше двух. Решает это один приём — стек, в котором мы сами помним, где остановились. Он же понадобится для JSON.stringify и JSON.parse.

Теория 1

Как работает настоящий flat

  1. depth — сколько уровней раскрыть. Не передан — 1. Превращается в целое: '2' → 2, 1.7 → 1, NaN → 0. Infinity — раскрыть всё.
  2. Раскрываются только настоящие массивы (Array.isArray), не строки и не «похожие на массив» объекты.
  3. Дырки выбрасываются на каждом уровне — даже при flat(0).
  4. Порядок элементов сохраняется: вложенный массив встаёт на своё место.

console.log([1, , [2, , 3]].flat(), [1, , 3].flat(0));    // [ 1, 2, 3 ] [ 1, 3 ]
console.log(['ab', { length: 1, 0: 'x' }, [[]]].flat());  // [ 'ab', { '0': 'x', length: 1 }, [] ]

Теория 2

Без рекурсии: стек кадров

Рекурсивный flat запоминает «где я остановился во внешнем массиве» за нас — в стеке вызовов. Без рекурсии этот стек заводим сами: массив кадров. Кадр — это «какой массив обходим, на каком индексе остановились и сколько уровней ещё можно раскрыть».

Один цикл:

  1. Смотрим на верхний кадр. Если он дошёл до конца — снимаем его и продолжаем внешний.
  2. Иначе берём его следующий элемент. Массив и можно раскрывать — кладём новый кадр сверху. Нет — пишем элемент в ответ.

Внутренний массив доходит до конца раньше, чем мы вернёмся к внешнему, — поэтому порядок элементов правильный.

const stack = [{ items: [1, [2, [3]], 4], i: 0 }];
const out = [];
while (stack.length > 0) {
  const top = stack[stack.length - 1];
  if (top.i === top.items.length) {
    stack.pop();
    continue;
  }
  const el = top.items[top.i++];
  if (Array.isArray(el)) stack.push({ items: el, i: 0 });
  else out.push(el);
  console.log('стек:', stack.length, 'результат:', out.join(','));
}

function myFlat(arr, depth) {
  let d = depth === undefined ? 1 : Math.trunc(Number(depth));
  if (d !== d) d = 0;                                  // NaN → 0

  const result = [];
  const stack = [{ items: arr, i: 0, depth: d }];     // кадр: массив, позиция, запас глубины

  while (stack.length > 0) {
    const top = stack[stack.length - 1];
    if (top.i >= top.items.length) {                   // массив закончился — вернуться к внешнему
      stack.pop();
      continue;
    }
    const i = top.i++;
    if (!(i in top.items)) continue;                   // дырка
    const el = top.items[i];
    if (Array.isArray(el) && top.depth > 0) {
      stack.push({ items: el, i: 0, depth: top.depth - 1 });
    } else {
      result.push(el);
    }
  }
  return result;
}

console.log(myFlat([1, [2, [3, [4]]]], 2)); // [1, 2, 3, [4]]
Разбор

Это стек кадров из теории плюс два правила настоящего flat:

  1. Глубина. Каждый кадр помнит свой запас: вложенный массив раскрываем, только если у текущего кадра depth > 0, и новому кадру даём на один меньше. Infinity - 1 — всё ещё Infinity, поэтому «всё» работает само. Math.trunc(Number(depth)) превращает '2' в 2 и 1.7 в 1, а NaN приходится ловить отдельно — это единственное значение, для которого d !== d.
  2. Дырки. Проверка i in top.items стоит до всего остального — дырка не попадает в ответ ни на каком уровне, даже при depth = 0.

Цикл один — while. Рекурсии нет: вложенность хранится в массиве stack, а не в вызовах функции.

Другой известный способ — «стек значений»: положить все элементы в стек, снимать по одному, массивы раскладывать обратно в стек, а в конце перевернуть ответ. Он короче, но там прячутся лишние циклы (push(...el) и reverse() сами обходят массив), и сложнее учесть глубину и дырки.

function flatDepth(arr) {
  let max = 1;
  const stack = [{ items: arr, level: 1 }];
  while (stack.length > 0) {
    const { items, level } = stack.pop();
    if (level > max) max = level;
    for (const el of items) {
      if (Array.isArray(el)) stack.push({ items: el, level: level + 1 });
    }
  }
  return max;
}

function flattenObject(obj) {
  const result = {};
  const stack = [{ obj, keys: Object.keys(obj), i: 0, prefix: '' }];

  while (stack.length > 0) {
    const top = stack[stack.length - 1];
    if (top.i >= top.keys.length) {
      stack.pop();
      continue;
    }
    const key = top.keys[top.i++];
    const value = top.obj[key];
    const path = top.prefix + key;

    const isNested = typeof value === 'object' && value !== null &&
      !Array.isArray(value) && Object.keys(value).length > 0;
    if (isNested) {
      stack.push({ obj: value, keys: Object.keys(value), i: 0, prefix: path + '.' });
    } else {
      result[path] = value;
    }
  }
  return result;
}

Напиши bubbleSort(arr) — сортировку пузырьком: числа по возрастанию, на месте, вернуть тот же массив. Если за проход не было ни одного обмена — массив уже отсортирован, выходим.

function bubbleSort(arr) {
  for (let end = arr.length - 1; end > 0; end--) {
    let swapped = false;
    for (let i = 0; i < end; i++) {
      if (arr[i] > arr[i + 1]) {
        [arr[i], arr[i + 1]] = [arr[i + 1], arr[i]];   // обмен соседей
        swapped = true;
      }
    }
    if (!swapped) break;
  }
  return arr;
}

function mySort(arr, compareFn) {
  if (compareFn !== undefined && typeof compareFn !== 'function') {
    throw new TypeError('The comparison function must be either a function or undefined');
  }

  // > 0 — значит, a должен стоять после b
  function order(a, b) {
    if (a === undefined) return b === undefined ? 0 : 1;
    if (b === undefined) return -1;
    if (compareFn) {
      const r = Number(compareFn(a, b));
      return r !== r ? 0 : r;              // NaN считается нулём
    }
    const x = String(a);
    const y = String(b);
    return x < y ? -1 : x > y ? 1 : 0;
  }

  const len = arr.length;
  let k = 0;                               // сколько элементов уже стоит отсортированными в начале
  for (let i = 0; i < len; i++) {
    if (!(i in arr)) continue;             // дырку не вставляем — в конце на её месте останется пустота
    const cur = arr[i];
    if (k < i) delete arr[i];              // префикс короче i — эта ячейка освобождается
    let j = k - 1;
    while (j >= 0 && order(arr[j], cur) > 0) {
      arr[j + 1] = arr[j];
      j--;
    }
    arr[j + 1] = cur;
    k++;
  }
  return arr;
}

Основа — сортировка вставками (два цикла: for и while). Всё остальное — правила настоящего sort, собранные в функцию order(a, b):

  2. Без compareFn — сравнение строк; < для строк сравнивает коды символов, как и настоящий sort.
  3. NaN в ответе функции — ноль: r !== r верно только для NaN.

Самое хитрое — дырки, и они решены без третьего цикла. Отсортированная часть живёт в начале массива, её длина — k. Дырку просто пропускаем, поэтому k может отставать от i. Когда отстаёт, ячейку i после чтения освобождаем delete — её значение сейчас переедет в отсортированную часть левее. К концу все ячейки от k до конца пустые: дырки оказались в самом конце, как у настоящего sort.

Устойчивость — из строгого сравнения > 0, как в insertionSort.

function escapeJSON(str) {
  let res = '"';
  for (let i = 0; i < str.length; i++) {
    const ch = str[i];
    const code = str.charCodeAt(i);
    if (ch === '"') res += '\\"';
    else if (ch === '\\') res += '\\\\';
    else if (ch === '\n') res += '\\n';
    else if (ch === '\r') res += '\\r';
    else if (ch === '\t') res += '\\t';
    else if (ch === '\b') res += '\\b';
    else if (ch === '\f') res += '\\f';
    else if (code < 32) res += '\\u00' + (code < 16 ? '0' : '') + code.toString(16);
    else res += ch;
  }
  return res + '"';
}

Теория 2

stringify без рекурсии: стек кадров

Вложенность решаем так же, как в myFlat (блок 5): стек кадров. Кадр — открытый массив или объект: что обходим, его ключи, на каком мы ключе и сколько уже записали (для запятых).

  • Встретили массив или объект — пишем [ или { и кладём новый кадр.
  • Кадр дошёл до конца — пишем ] или } и снимаем его.
  • Иначе — берём следующий ключ и пишем значение.

Цикл один — по стеку. Второй — внутри экранирования строк. Рекурсии нет.

Циклическая ссылка: объекты, которые сейчас открыты, лежат в Set. Если открываем объект, который уже там, — мы в цикле, TypeError. При закрытии кадра объект из Set удаляем: один и тот же объект в двух разных местах ([obj, obj]) — не цикл.

function myStringify(value) {
  const seen = new Set();   // объекты, которые сейчас открыты, — чтобы поймать цикл
  const stack = [];         // кадры: открытые массивы и объекты
  let out = '';

  function quote(str) {     // цикл № 1: экранирование строки
    let res = '"';
    for (let i = 0; i < str.length; i++) {
      const ch = str[i];
      const code = str.charCodeAt(i);
      if (ch === '"') res += '\\"';
      else if (ch === '\\') res += '\\\\';
      else if (ch === '\n') res += '\\n';
      else if (ch === '\r') res += '\\r';
      else if (ch === '\t') res += '\\t';
      else if (ch === '\b') res += '\\b';
      else if (ch === '\f') res += '\\f';
      else if (code < 32) res += '\\u00' + (code < 16 ? '0' : '') + code.toString(16);
      else res += ch;
    }
    return res + '"';
  }

  // если у объекта есть toJSON (как у Date) — пишем его результат
  function prepare(v, key) {
    if (v !== null && typeof v === 'object' && typeof v.toJSON === 'function') return v.toJSON(key);
    return v;
  }

  // пишет значение; массив или объект — открывает кадром. false — такое значение не пишется
  function emit(v) {
    if (v === null) { out += 'null'; return true; }
    const type = typeof v;
    if (type === 'string') { out += quote(v); return true; }
    if (type === 'number') { out += v === v && v !== Infinity && v !== -Infinity ? String(v) : 'null'; return true; }
    if (type === 'boolean') { out += v ? 'true' : 'false'; return true; }
    if (type === 'bigint') throw new TypeError('Do not know how to serialize a BigInt');
    if (type === 'object') {
      if (seen.has(v)) throw new TypeError('Converting circular structure to JSON');
      seen.add(v);
      const isArray = Array.isArray(v);
      stack.push({ value: v, isArray, keys: isArray ? null : Object.keys(v), i: 0, count: 0 });
      out += isArray ? '[' : '{';
      return true;
    }
    return false;             // undefined, функция, символ
  }

  if (!emit(prepare(value, ''))) return undefined;

  while (stack.length > 0) {  // цикл № 2: по стеку кадров
    const top = stack[stack.length - 1];
    const len = top.isArray ? top.value.length : top.keys.length;
    if (top.i >= len) {                       // кадр закончился — закрыть скобку
      out += top.isArray ? ']' : '}';
      seen.delete(top.value);
      stack.pop();
      continue;
    }
    const key = top.isArray ? top.i : top.keys[top.i];
    top.i++;
    const v = prepare(top.value[key], String(key));

    if (top.isArray) {
      if (top.count > 0) out += ',';
      top.count++;
      if (!emit(v)) out += 'null';            // в массиве «непишущееся» — null
    } else {
      const type = typeof v;
      if (v === undefined || type === 'function' || type === 'symbol') continue;   // в объекте — пропустить
      if (top.count > 0) out += ',';
      top.count++;
      out += quote(key) + ':';
      emit(v);
    }
  }
  return out;
}

Решение — три маленькие функции без циклов вокруг двух циклов:

  1. quote — экранирование из escapeJSON (цикл № 1).
  2. prepare — правило toJSON: если у объекта есть такой метод, пишем то, что он вернул. Так Date становится строкой.
  3. emit — пишет одно значение по таблице из теории. Примитив — сразу текстом. Массив или объект — пишет открывающую скобку и кладёт кадр в стек; дальше им займётся главный цикл. «Непишущееся» (undefined, функция, символ) — возвращает false, и решение принимает вызвавший: в массиве это null, в объекте ключ выбрасывается.

Главный цикл (№ 2) каждый шаг смотрит на верхний кадр: кончились ключи — закрыть скобку; нет — записать следующий элемент. Запятая ставится по счётчику записанных count, а не по индексу — из-за пропущенных ключей.

Порядок ключей даёт Object.keys: сначала целые числа по возрастанию, потом строки в порядке добавления — так же, как у настоящего JSON.stringify.

Чего нет: replacer, space, объекты-обёртки вроде new String('x') и «одинокие» половинки эмодзи (их настоящий метод пишет как \ud83d). Это редкость — на экзамене достаточно сказать про них словами.

  if (c >= 'a' && c <= 'f') return c.charCodeAt(0) - 87;

function hex4(text, pos) {               // четыре шестнадцатеричные цифры -> код, иначе -1
  const a = hexDigit(text[pos]);
  const b = hexDigit(text[pos + 1]);
  const c = hexDigit(text[pos + 2]);
  const d = hexDigit(text[pos + 3]);
  if (a < 0 || b < 0 || c < 0 || d < 0) return -1;
  return a * 4096 + b * 256 + c * 16 + d;
}

function parseJSONString(text) {
  if (text[0] !== '"') throw new SyntaxError('Expected "');
  let result = '';
  for (let j = 1; j < text.length; j++) {
    const c = text[j];
    if (c === '"') {
      if (j !== text.length - 1) throw new SyntaxError('Unexpected text after string');
      return result;
    }
    if (c === '\\') {
      const esc = text[j + 1];
      if (esc === '"' || esc === '\\' || esc === '/') result += esc;
      else if (esc === 'n') result += '\n';
      else if (esc === 't') result += '\t';
      else if (esc === 'r') result += '\r';
      else if (esc === 'b') result += '\b';
      else if (esc === 'f') result += '\f';
      else if (esc === 'u') {
        const code = hex4(text, j + 2);
        if (code < 0) throw new SyntaxError('Bad Unicode escape');
        result += String.fromCharCode(code);
        j += 4;
      } else {
        throw new SyntaxError('Bad escaped character');
      }
      j++;                                // пропустить букву после слэша
      continue;
    }
    if (c < ' ') throw new SyntaxError('Bad control character in string literal');
    result += c;
  }
  throw new SyntaxError('Unterminated string');
}

function nextNumState(state, c) {
  const digit = c >= '0' && c <= '9';
  const e = c === 'e' || c === 'E';
  if (state === 'start') return c === '-' ? 'minus' : c === '0' ? 'zero' : digit ? 'int' : 'bad';
  if (state === 'minus') return c === '0' ? 'zero' : digit ? 'int' : 'bad';
  if (state === 'zero') return c === '.' ? 'dot' : e ? 'e' : 'bad';
  if (state === 'int') return digit ? 'int' : c === '.' ? 'dot' : e ? 'e' : 'bad';
  if (state === 'dot') return digit ? 'frac' : 'bad';
  if (state === 'frac') return digit ? 'frac' : e ? 'e' : 'bad';
  if (state === 'e') return c === '+' || c === '-' ? 'esign' : digit ? 'exp' : 'bad';
  if (state === 'esign' || state === 'exp') return digit ? 'exp' : 'bad';
  return 'bad';
}

function parseNumberJSON(text) {
  let state = 'start';
  for (let i = 0; i < text.length; i++) state = nextNumState(state, text[i]);
  const finished = state === 'zero' || state === 'int' || state === 'frac' || state === 'exp';
  if (!finished) throw new SyntaxError('Bad number: ' + text);
  return Number(text);
}

  function fail(msg) {
    throw new SyntaxError(msg + ' in JSON at position ' + i);
  }

  // положить готовое значение туда, где его сейчас ждут
  function put(v) {
    if (stack.length === 0) {
      if (hasResult) fail('Unexpected non-whitespace character after JSON');
      result = v;
      hasResult = true;
      return;
    }
    const top = stack[stack.length - 1];
    if (top.isArray) {
      if (top.state !== 'first' && top.state !== 'value') fail('Unexpected value');
      top.value.push(v);
    } else {
      if (top.state !== 'value') fail('Unexpected value');
      top.value[top.key] = v;
    }
    top.state = 'next';
  }

    const ch = text[i];
    if (ch === ' ' || ch === '\n' || ch === '\r' || ch === '\t') { i++; continue; }
    const top = stack[stack.length - 1];

    if (ch === '{' || ch === '[') {
      const container = ch === '[' ? [] : {};
      put(container);
      stack.push({ value: container, isArray: ch === '[', state: 'first', key: null });
      i++;

      i++;
      continue;

      i++;
      continue;

    if (ch === ':') {
      if (!top || top.state !== 'colon') fail('Unexpected token :');
      top.state = 'value';
      i++;
      continue;

          else if (esc === 't') buf += '\t';
          else if (esc === 'r') buf += '\r';

            buf += String.fromCharCode(code);
            j += 4;
          } else { i = j; fail('Bad escaped character'); }
          j++;
          continue;

        if (c < ' ') { i = j; fail('Bad control character in string literal'); }
        buf += c;
      } else {
        if (c === ',' || c === ']' || c === '}' || c === ':' || c === ' ' || c === '\n' || c === '\r' || c === '\t') break;

    if (isString) {
      if (!closed) fail('Unterminated string');
      i = j + 1;
      if (top && !top.isArray && (top.state === 'first' || top.state === 'key')) {
        top.key = buf;                            // строка на месте ключа
        top.state = 'colon';
      } else {
        put(buf);
      }
      continue;
    }

    i = j;
    if (buf === 'true') put(true);
    else if (buf === 'false') put(false);
    else if (buf === 'null') put(null);
    else if (num === 'zero' || num === 'int' || num === 'frac' || num === 'exp') put(Number(buf));
    else fail('Unexpected token ' + buf[0]);
  }

  if (!hasResult || stack.length > 0) fail('Unexpected end of JSON input');
  return result;
}

console.log(myParse('{"a":[1,{"b":"x"}],"ok":true}'));
Разбор

Два цикла по ролям:

  1. Внешний идёт по символам и обрабатывает структуру: скобки, запятые, двоеточия. Он же ведёт стек кадров, у каждого — состояние из таблицы в теории.
  2. Внутренний читает один атом. Для строки он разбирает экранирование (как parseJSONString), для слова — копит символы и параллельно гонит автомат числа (как parseNumberJSON). Одна и та же переменная цикла обслуживает оба режима — поэтому циклов два, а не три.

Всё готовое значение проходит через одну функцию put: она знает, куда его положить (в корень, в массив или в объект под последним ключом) и проверяет, ждали ли здесь значение. Именно так ловятся ошибки вроде [1 2] (после значения массив ждёт , или ]) и {"a" 1} (после ключа ждём :).

Строка — единственный атом, который бывает и ключом, и значением. Отличаем по состоянию объекта: если он ждёт ключ (first или key), строка становится ключом.

В конце: результат должен быть, а все скобки — закрыты; иначе текст оборвался. Лишний текст после значения ловит тот же put: корень уже заполнен.

Рекурсии нет: вложенность — это стек кадров, как в myFlat и myStringify. Три задачи лекции «без рекурсии» решены одним и тем же приёмом.

function parseDate(text) {
  if (!/^\d\d\.\d\d\.\d{4}$/.test(text)) return null;
  const [day, month, year] = text.split('.').map(Number);
  const date = new Date(year, month - 1, day);

  const same = date.getFullYear() === year && date.getMonth() === month - 1 && date.getDate() === day;
  return same ? date : null;
}

function countWeekdays(from, to) {

Массив в V8: виды элементов и ёмкость

По стандарту массив — это обычный объект с ключами '0', '1', … и особым свойством length. Но V8 (движок в Chrome и Node) хранит элементы отдельно от остальных свойств — сплошным блоком памяти, как в C. Поэтому чтение arr[i] — это O(1).

Виды элементов (elements kinds). V8 запоминает, что лежит в массиве, и выбирает самый экономный способ хранения:

  • PACKED_SMI_ELEMENTS — только маленькие целые числа: [1, 2, 3];
  • PACKED_DOUBLE_ELEMENTS — есть дробные: [1, 2, 3.5] (числа хранятся «как есть», без упаковки);
  • PACKED_ELEMENTS — что угодно: строки, объекты;
  • HOLEY_… — те же три вида, но с дырками: new Array(3), arr[10] = 1 при длине 3.

Переходы идут только в одну сторону — к более общему виду: добавили в [1, 2, 3] строку и потом удалили её — массив всё равно останется PACKED_ELEMENTS; заполнили дырку — останется HOLEY.

Ёмкость. Блок памяти выделяется с запасом. Когда место кончается, V8 выделяет новый блок размером n + n/2 + 16 (где n — нужная длина) и копирует туда элементы. Для пустого массива, в который делают push, ёмкость растёт так: 17 → 43 → 82 → 140… Поэтому push в среднем O(1): дорогие копирования случаются редко.

Словарный режим. Если массив очень разреженный (arr[100000000] = 1), выделять сто миллионов ячеек нет смысла — V8 переключает его в DICTIONARY_ELEMENTS, то есть в хеш-таблицу. Это медленнее.

Ответ

Элементы массива лежат сплошным блоком по индексам. pop() убирает последний элемент — остальные остаются на своих местах, это одна операция, O(1).

Напиши createDynamicArray() — свой «массив с ёмкостью», как в V8. Внутри — хранилище фиксированного размера. Методы: push(value) (вернуть новую длину), get(i); свойства length, capacity (размер хранилища) и copies (сколько элементов пришлось переписать при всех расширениях). Когда место кончилось, новое хранилище — размером n + n/2 + 16 (округление вниз), где n — новая длина.

После первого push ёмкость 17, после 18-го — 43.

Показать кодСкрыть
function createDynamicArray() {
  let store = new Array(0);   // «блок памяти»: его размер не меняем, только заменяем целиком
  let length = 0;
  let copies = 0;

  return {
    push(value) {
      if (length === store.length) {                       // места нет
        const n = length + 1;
        const bigger = new Array(n + (n >> 1) + 16);       // n + n/2 + 16
        for (let i = 0; i < length; i++) bigger[i] = store[i];
        copies += length;
        store = bigger;
      }
      store[length] = value;
      length++;
      return length;
    },
    get(i) {
      return i >= 0 && i < length ? store[i] : undefined;
    },
    get length() { return length; },
    get capacity() { return store.length; },
    get copies() { return copies; },
  };
}

const arr = createDynamicArray();
arr.push('a');
console.log(arr.length, arr.capacity); // 1 17
Разбор

Хранилище создаётся один раз заданного размера и дальше не растёт — как блок памяти в C. Пока есть место, push просто пишет в следующую ячейку. Когда место кончилось — выделяем новое хранилище побольше и переписываем всё туда: это дорого, O(n).

Смысл запаса — в последних тестах: за 1000 push расширений было всего девять (ёмкость 17, 43, 82, 140, 227, 358, 554, 848, 1289), и переписали мы 2269 элементов — в среднем меньше трёх копирований на один push. При росте в полтора раза так будет всегда: суммарно копий не больше чем примерно 3n. Поэтому говорят «push — O(1) амортизированно»: изредка дорого, в среднем дёшево. Если бы ёмкость росла на 1, каждый push копировал бы весь массив, и 1000 push стоили бы полмиллиона копирований.

n >> 1 — целое деление на 2 сдвигом битов (из блока операторов первого файла). Геттеры get length() делают свойства только для чтения снаружи.

Теория 2

Set и Map: хеш-таблица с порядком

Стандарт требует, чтобы Map и Set находили элемент быстрее, чем перебором (в среднем). На практике это хеш-таблица:

  1. По ключу считается число — хеш. Остаток от деления хеша на число корзин — номер корзины.
  2. При поиске смотрим только одну корзину, а не всё подряд — в среднем O(1).
  3. Разные ключи иногда попадают в одну корзину (коллизия) — поэтому в корзине список, и ключи в нём сравниваются по-настоящему.
  4. Когда записей становится много, корзин делают вдвое больше и всё раскладывают заново (rehash).

В V8 это OrderedHashMap / OrderedHashSet (их видно в %DebugPrint(new Map())) — «детерминированная хеш-таблица»: записи лежат в отдельном массиве в порядке добавления, а корзины хранят номера записей. Удаление помечает запись как удалённую (дырка), при следующем rehash дырки выбрасываются. Отсюда главное свойство Map и Set: перебор всегда в порядке добавления.

Ключи сравниваются по правилу SameValueZero (как includes): NaN равен NaN, объекты — по ссылке. Хеш строки считается из её символов, а объекту V8 при первой надобности выдаёт случайное число и хранит его вместе с объектом.

  2. Set выбросил повторы и сохранил порядок первого появления.
  3. JSON.stringify не умеет Map — у него нет своих перечислимых ключей, получается {}. Чтобы сохранить Map, его превращают в массив пар: [...m].

Напиши hashString(str, size) — номер корзины от 0 до size - 1: полиномиальный хеш h = h * 31 + код символа, после каждого шага — остаток от деления на size.

Разбор

Это та же схема Горнера, что и «число из цифр» в первом файле: h * основание + цифра. Только «цифры» — коды символов, основание — 31, и чтобы число не росло бесконечно, после каждого шага берём остаток.

Хорошая хеш-функция раскладывает разные строки по разным корзинам равномерно и учитывает порядок символов: 'ab' и 'ba' попадают в разные корзины (а сумма кодов у них одинаковая). Одна и та же строка всегда даёт один и тот же хеш — иначе мы не нашли бы её при поиске.

6Сложно★ минимум

Напиши createHashMap() — свой Map на хеш-таблице, устроенный как в V8: записи в массиве в порядке добавления, корзины хранят номера записей, удаление оставляет «дырку», при переполнении — вдвое больше корзин и перестройка без дырок. Методы set (возвращает сам объект), get, has, delete (true/false), свойство size, метод keys() — массив ключей по порядку. Ключи сравниваются по SameValueZero; 1 и '1' — разные ключи.

Показать кодСкрыть
function hashString(str, size) {
  let h = 0;
  for (let i = 0; i < str.length; i++) h = (h * 31 + str.charCodeAt(i)) % size;
  return h;
}

function sameValueZero(a, b) {
  return a === b || (a !== a && b !== b);
}

function createHashMap() {
  let buckets = [[], [], [], []];   // корзины: в каждой — номера записей
  let entries = [];                 // записи в порядке добавления
  let size = 0;

  // тип входит в хеш, чтобы 1 и '1' попадали в разные корзины
  function bucketOf(key, count) {
    return hashString(typeof key + ':' + String(key), count);
  }

  function find(key) {              // номер живой записи или -1
    for (const index of buckets[bucketOf(key, buckets.length)]) {
      const e = entries[index];
      if (!e.deleted && sameValueZero(e.key, key)) return index;
    }
    return -1;
  }

  function rehash() {               // корзин вдвое больше, удалённые записи выбрасываем
    const live = entries.filter(e => !e.deleted);
    buckets = Array.from({ length: buckets.length * 2 }, () => []);
    entries = [];
    for (const e of live) {
      entries.push(e);
      buckets[bucketOf(e.key, buckets.length)].push(entries.length - 1);
    }
  }

  return {
    set(key, value) {
      const index = find(key);
      if (index !== -1) {
        entries[index].value = value;          // ключ уже есть — место в порядке не меняется
        return this;
      }
      if (entries.length + 1 > buckets.length * 2) rehash();   // в среднем не больше 2 записей на корзину
      entries.push({ key, value, deleted: false });
      buckets[bucketOf(key, buckets.length)].push(entries.length - 1);
      size++;
      return this;
    },
    get(key) {
      const index = find(key);
      return index === -1 ? undefined : entries[index].value;
    },
    has(key) {
      return find(key) !== -1;
    },
    delete(key) {
      const index = find(key);
      if (index === -1) return false;
      entries[index].deleted = true;           // дырка: порядок остальных не сдвигается
      size--;
      return true;
    },
    get size() {
      return size;
    },
    keys() {
      return entries.filter(e => !e.deleted).map(e => e.key);
    },
  };
}

const map = createHashMap().set('a', 1).set('b', 2);
console.log(map.get('b'), map.size, map.keys()); // 2 2 [ 'a', 'b' ]
Разбор

Две структуры, как в V8:

  • entries — все записи в порядке добавления; отсюда порядок в keys();
  • buckets — корзины; в корзине не сами записи, а их номера в entries.

Поиск (find): считаем корзину по хешу, перебираем только номера в ней и сравниваем ключи по SameValueZero. В среднем в корзине 1–2 записи — поэтому O(1), даже когда записей тысяча.

Удаление не сдвигает массив (это было бы O(n)), а ставит пометку deleted — «дырку». find дырки пропускает, keys() — тоже. Повторное добавление удалённого ключа кладёт его в конец — как у настоящего Map (тест ['b', 'c', 'a']). А set для существующего ключа меняет только значение — место не меняется.

Перестройка (rehash): когда записей (вместе с дырками) стало вдвое больше, чем корзин, корзин делаем вдвое больше и раскладываем живые записи заново — хеш зависит от числа корзин, поэтому старые номера корзин уже не годятся. Дырки при этом исчезают.

Слабое место: все объекты-ключи дают строку 'object:[object Object]' и попадают в одну корзину — работает правильно (сравнение по ссылке), но медленно. V8 решает это, выдавая каждому объекту скрытое случайное число-хеш.

Set — это та же таблица, только без значений.

Напиши createLRUCache(capacity) — кеш на capacity записей: get(key) (или undefined) и set(key, value). Если места нет, выбрасывается запись, к которой дольше всего не обращались (и get, и set считаются обращением). Используй то, что Map хранит порядок добавления.

Показать кодСкрыть
function createLRUCache(capacity) {
  const map = new Map();          // порядок: от давно использованных к недавним
  return {
    get(key) {
      if (!map.has(key)) return undefined;
      const value = map.get(key);
      map.delete(key);            // переставить в конец: теперь это самая свежая запись
      map.set(key, value);
      return value;
    },
    set(key, value) {
      if (map.has(key)) map.delete(key);
      map.set(key, value);
      if (map.size > capacity) {
        const oldest = map.keys().next().value;   // первый ключ — самый давний
        map.delete(oldest);
      }
    },
    keys() {
      return [...map.keys()];
    },
  };
}

const cache = createLRUCache(2);
cache.set('a', 1);
cache.set('b', 2);
cache.get('a');
cache.set('c', 3);
console.log(cache.keys()); // ['a', 'c']
Разбор

Весь «алгоритм» — в свойстве Map: перебор идёт в порядке добавления. Если при каждом обращении удалить и добавить ключ заново, он уедет в конец. Тогда в начале всегда самый давно использованный — его и выбрасываем.

map.keys() возвращает итератор; .next().value — его первый элемент, без создания массива всех ключей.

Это классическая задача с собеседований — и хороший ответ на вопрос «зачем Map, если есть объект»: объект сортирует числовые ключи и не хранит порядок для них.

Теория 1

Сборщик мусора: что считается мусором

В JS память не освобождают вручную — это делает сборщик мусора (garbage collector, GC). Мусор — это объект, до которого нельзя добраться из программы.

Откуда «добираются» — это корни: глобальный объект, переменные функций, которые сейчас выполняются (стек вызовов), замыкания этих функций. Объект достижим, если до него есть цепочка ссылок от корня. Остальное — мусор.

Основной алгоритм — mark-and-sweep («пометить и вымести»):

  1. Mark — от корней обойти все ссылки и пометить каждый достижимый объект.
  2. Sweep — всё непомеченное считается свободной памятью.

Поэтому циклические ссылки не мешают: два объекта, ссылающиеся друг на друга, но недостижимые из корней, удалятся оба. (В старом алгоритме «подсчёт ссылок» они жили бы вечно.)

let user = { name: 'Kate' };   // объект достижим через переменную user
let admin = user;              // ещё одна ссылка
user = null;                   // объект всё ещё достижим через admin
admin = null;                  // ссылок нет — объект стал мусором и будет удалён

Утечка памяти в JS — это не «забыли удалить», а «случайно оставили достижимым»: глобальная переменная, забытый setInterval, обработчик события, который держит большой объект в замыкании, кеш (Map), который только растёт.

Garbage collector — часть движка, которая сама освобождает память от объектов, которые программе больше не нужны. Программист память не освобождает.

Мусор — объект, недостижимый из корней: глобального объекта, переменных выполняющихся сейчас функций и их замыканий. Сам факт «на объект кто-то ссылается» не спасает — важно, есть ли путь от корня.

Теория 2

Как устроен сборщик в V8 и сколько он стоит

В V8 сборщик называется Orinoco. Он опирается на наблюдение: большинство объектов умирают молодыми (временные строки, объекты внутри функций). Поэтому память разделена на поколения:

  • Молодое поколение — маленькое. Его чистит быстрый сборщик Scavenger: живые объекты копируются в другую половину, всё остальное просто считается свободным. Объекты, пережившие две такие чистки, переезжают в старое поколение.
  • Старое поколение — большое. Его чистит Mark-Compact: пометить живые, вымести мёртвые, уплотнить память, чтобы не было дыр. Большая часть этой работы идёт параллельно в фоновых потоках и по кусочкам (incremental), чтобы главный поток не стоял долго.

Как часто запускается? Не по таймеру, а когда кончается место: молодое поколение заполнилось — быстрая чистка (в логе причина так и пишется: allocation failure); старое выросло за подобранный движком порог — большая. Браузер ещё пользуется моментами простоя.

Сколько стоит? Замер на Node 26 (node --trace-gc): программа создала 3 миллиона временных объектов — прошло 56 быстрых чисток, каждая 0,17 мс в медиане (максимум ~1 мс). Программа, которая удерживает миллионы объектов, вызвала большие чистки старого поколения, и главный поток простоял 30 и 150 мс (кучи 66 и 280 МБ): пометка шла по кусочкам, но завершающая пауза на большой куче всё равно заметна. Обычно сборщик занимает несколько процентов времени процессора; по памяти — молодому поколению нужно две половины для копирования, а вся куча Node ограничена (здесь v8.getHeapStatistics().heap_size_limit показал ~4 ГБ; меняется флагом --max-old-space-size).

Выключить нельзя. Из JS сборщик не отключается (есть только флаг --expose-gc, чтобы вызвать его вручную через global.gc()). Если бы память не освобождалась, куча росла бы до лимита, и процесс упал бы с ошибкой FATAL ERROR: Reached heap limit Allocation failed - JavaScript heap out of memory. В языках без сборщика (C) память освобождают вручную — отсюда утечки и ошибки «использовали уже освобождённую память».

Теория 3

WeakMap, WeakSet, WeakRef

Обычный Map держит ключ сильно: пока объект лежит ключом в Map, он достижим и не удаляется — даже если больше нигде не нужен. Так кеши и становятся утечками.

  • WeakMap — ключи только объекты (и символы, не созданные через Symbol.for), и держит их слабо: если на объект-ключ больше никто не ссылается, запись исчезает вместе с ним. Поэтому у WeakMap нет size, перебора и clear — содержимое зависит от того, когда прошёл сборщик. Методы: set, get, has, delete.
  • WeakSet — то же для набора объектов: add, has, delete. «Пометить объект, не удерживая его».
  • WeakRef — слабая ссылка на один объект: ref.deref() возвращает объект или undefined, если его уже удалили.
  • FinalizationRegistry — колбэк «объект удалён». Когда он вызовется и вызовется ли вообще — не гарантируется; строить на нём логику программы нельзя.

function memoizeByObject(fn) {
  const cache = new WeakMap();
  return function (obj) {
    if (cache.has(obj)) return cache.get(obj);
    const result = fn(obj);
    cache.set(obj, result);
    return result;
  };
}

function deepCloneCycles(value, copies = new WeakMap()) {
  if (value === null || typeof value !== 'object') return value;
  if (copies.has(value)) return copies.get(value);   // уже копировали — отдать ту же копию

  const copy = Array.isArray(value) ? [] : {};
  copies.set(value, copy);                             // запомнить ДО обхода детей
  for (const key of Object.keys(value)) {
    copy[key] = deepCloneCycles(value[key], copies);
  }
  return copy;
}

Разбор

Ответ

Ключевая роль — хранить что-то про объект, не мешая сборщику этот объект удалить. Обычный Map или Set держит свои ключи сильно: объект-ключ достижим через коллекцию и никогда не станет мусором.

  • WeakMap — «данные при объекте»: кеш результатов по объекту-аргументу (memoizeByObject), приватные данные экземпляра, метаданные DOM-узла (узел удалили со страницы — данные исчезли сами), словарь «оригинал → копия» при глубоком копировании.
  • WeakSet — «пометка»: объект уже обработан, уже проверен, уже посещён при обходе.
  • WeakRef — ссылка «если объект ещё жив»: большой кеш (картинки, результаты вычислений), который не жалко потерять при нехватке памяти.
  • FinalizationRegistry — уведомление, что объект удалён (например, освободить связанный внешний ресурс), но без гарантий времени.

Цена слабости: у WeakMap/WeakSet нет size и перебора, ключи — только объекты (и незарегистрированные символы). Содержимое зависит от сборщика, поэтому узнать «что сейчас внутри» нельзя.

Теория 4

Какие данные сортируются сами, а какие хранят порядок

Сами по значению не сортирует никто. В JS нет встроенной «сортированной коллекции»: чтобы держать данные упорядоченными, их сортируют вручную или вставляют на нужное место.

Но у обычного объекта есть правило порядка ключей (оно же для Object.keys, for...in, JSON.stringify, Object.entries):

  1. сначала целые ключи-индексы ('0', '2', '10') — по возрастанию;
  2. потом остальные строки ('b', '-1', '1.5', '01') — в порядке добавления;
  3. потом символы — в порядке добавления (их видно только через Reflect.ownKeys).

Порядок добавления хранят: Map и Set — для любых ключей, строковые ключи объекта. Повторный set существующего ключа место не меняет; удалить и добавить заново — ключ уходит в конец.

Массив — порядок индексов, то есть тот, который задал ты.

Напиши insertSorted(arr, value) — вставить число в уже отсортированный по возрастанию массив так, чтобы порядок сохранился (равные — после существующих). Место ищи двоичным поиском, вставляй через splice. Вернуть тот же массив.

function insertSorted(arr, value) {
  let lo = 0;
  let hi = arr.length;              // ищем первое место, где элемент больше value
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (arr[mid] <= value) lo = mid + 1;
    else hi = mid;
  }
  arr.splice(lo, 0, value);
  return arr;
}

console.log(insertSorted([1, 3, 5], 4)); // [1, 3, 4, 5]
Разбор

Раз встроенных сортированных коллекций нет, порядок поддерживают сами. Сортировать весь массив после каждой вставки — O(n log n) на вставку. Двоичный поиск находит место за O(log n): каждый шаг отбрасывает половину — смотрим в середину, если там элемент не больше value, место правее, иначе — левее или здесь.

splice всё равно сдвигает хвост массива (O(n), как unshift), но сравнений намного меньше. В базах данных для этого используют деревья — там и поиск, и вставка O(log n).

function toIndex(value, len, fallback) {
  if (value === undefined) return fallback;
  const n = Math.trunc(value) || 0;                    // дробь отбрасываем, NaN → 0
  return n < 0 ? Math.max(len + n, 0) : Math.min(n, len);
}

function mySlice(arr, start, end) {
  const len = arr.length;
  const from = toIndex(start, len, 0);
  const to = toIndex(end, len, len);
  const result = [];
  let n = 0;
  for (let k = from; k < to; k++, n++) {
    if (k in arr) result[n] = arr[k];                  // дырку не заполняем
  }
  result.length = Math.max(to - from, 0);              // хвостовые дырки тоже считаются в длину
  return result;
}

console.log(mySlice([1, 2, 3, 4], 1, 3)); // [2, 3]
Разбор

Вся сложность slice — в превращении аргументов в индексы, поэтому она вынесена в toIndex:

  1. не передан — значение по умолчанию (0 для начала, длина для конца);
  2. Math.trunc отбрасывает дробь, || 0 превращает NaN в 0;
  3. отрицательный — считаем от конца, но не левее нуля; положительный — не дальше длины.

Дальше один цикл от from до to. Счётчик n — позиция в ответе. Дырки сохраняются: пропущенный индекс остаётся пустым, а длину ответа задаём явно — иначе дырка в самом конце ([1, 2, ,].slice(1) — длина 2) потерялась бы.

Если start больше end, цикл не выполнится ни разу, а Math.max(…, 0) не даст отрицательной длины.

  const result = {};
  for (const path of Object.keys(flat)) {
    const keys = path.split('.');
    let node = result;

}

Напиши sumInCents(str) — сумму всех денежных сумм в строке в целых копейках (центах), чтобы не было ошибок округления. Дробная часть — через точку или запятую (разделитель — только если после него цифра), цифры после второй дробной отбрасываются. Один for, charCodeAt, без parseFloat и массивов.

function sumInCents(str) {
  let whole = 0;          // целая часть текущего числа
  let frac = 0;           // дробная часть — не больше двух цифр
  let fracDigits = 0;
  let started = false;
  let afterSep = false;
  let total = 0;

  for (let i = 0; i <= str.length; i++) {
    const code = str.charCodeAt(i);
    const next = str.charCodeAt(i + 1);
    if (code >= 48 && code <= 57) {
      if (!afterSep) whole = whole * 10 + (code - 48);
      else if (fracDigits < 2) {
        frac = frac * 10 + (code - 48);
        fracDigits++;
      }
      started = true;
    } else if ((code === 44 || code === 46) && started && !afterSep && next >= 48 && next <= 57) {
      afterSep = true;
    } else if (started) {
      total += whole * 100 + (fracDigits === 1 ? frac * 10 : frac);   // '.5' — это 50 копеек
      whole = 0;
      frac = 0;
      fracDigits = 0;
      started = false;
      afterSep = false;
    }
  }
  return total;
}

console.log(sumInCents('12.5 and 0.1 and 0.2')); // 1280
Разбор

Приём из первого файла (toCents): деньги считаем в целых копейках — сложение целых не даёт погрешности, и 0.1 + 0.2 превращается в честные 10 + 20 = 30.

Целую и дробную части собираем в разные переменные. Одна цифра после разделителя — это десятки копеек ('.5' — 50), поэтому умножаем на 10; две — как есть. Третья и дальше просто не учитываются.

Остальное — скелет «все числа строки» из блока 1: флаг started, разделитель только перед цифрой, лишний шаг цикла i <= str.length.

Смесь двух задач прошлого экзамена: читая строку справа налево (ровно один обратный for), найди последние два числа и верни курс последнее / предпоследнее. Дробная часть — через точку или запятую. Меньше двух чисел — null. Без parseFloat, Number() и массивов.

function rateFromEnd(str) {
  let current = 0;       // число, которое читаем сейчас (задом наперёд)
  let power = 1;         // вес следующей цифры
  let scale = 1;         // делитель дроби текущего числа
  let count = 0;
  let last = 0;          // последнее число строки — первое, которое мы дочитали
  let lastScale = 1;

  for (let i = str.length - 1; i >= 0; i--) {
    const code = str.charCodeAt(i);
    if (code >= 48 && code <= 57) {
      current += (code - 48) * power;
      power *= 10;

      const left = str.charCodeAt(i - 1);
      const left2 = str.charCodeAt(i - 2);
      const leftIsDigit = left >= 48 && left <= 57;
      const leftIsSep = (left === 44 || left === 46) && left2 >= 48 && left2 <= 57 && scale === 1;

      if (!leftIsDigit && !leftIsSep) {            // число закончилось (слева не цифра)
        count++;
        if (count === 2) {
          // (last / lastScale) / (current / scale) — одним делением
          return (last * scale) / (current * lastScale);
        }
        last = current;
        lastScale = scale;
        current = 0;
        power = 1;
        scale = 1;
      }
    } else if ((code === 44 || code === 46) && power > 1 && scale === 1) {
      scale = power;                               // всё прочитанное было дробной частью
    }
  }
  return null;
}

console.log(rateFromEnd('Exchanged 250 USD to 23125 CZK')); // 92.5
Разбор

Две задачи прошлого экзамена в одной: чтение справа налево и конец числа «по левому соседу» — из contactNumber, курс одним делением — из parseCurrencyRate.

  1. Читая с конца, первым мы дочитываем последнее число строки — запоминаем его как last вместе с делителем.
  2. Вторым дочитываем предпоследнее — и сразу отвечаем: (last / lastScale) / (current / scale), переписанное как одно деление (last · scale) / (current · lastScale).
  3. Разделитель внутри числа (слева цифра) превращает всё уже прочитанное в дробную часть: scale = power. Условие scale === 1 не даёт второму разделителю испортить число — '1.2.3' справа читается как 2.3, как в lastFloat.

Строка разбирается не до конца: как только два числа найдены, дальше читать незачем.
